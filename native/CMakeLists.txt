    find_library(OPM_XCB_LIBRARY xcb REQUIRED)
    find_library(OPM_XINPUT_LIBRARY xcb-xinput REQUIRED)
    add_library(opm_input STATIC physical_keys.cpp physical_keys.hpp)
    set_target_properties(opm_input PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(opm_input PUBLIC Qt6::Gui PRIVATE ${OPM_XCB_LIBRARY} ${OPM_XINPUT_LIBRARY})
    target_include_directories(opm_input PRIVATE ${OPM_XCB_INCLUDE})
    target_include_directories(opm_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(opm_input PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(opm_native STATIC
    model.cpp applications.cpp interaction.cpp power.cpp launch.cpp storage.cpp layouts.cpp editor_session.cpp editor_session.hpp editor_launch.cpp catalog.qrc)
target_link_libraries(opm_native PUBLIC Qt6::Core Qt6::DBus)
if(KF6Config_FOUND)
    target_link_libraries(opm_native PUBLIC KF6::ConfigCore)
    target_compile_definitions(opm_native PRIVATE OPM_HAVE_KCONFIG=1)
else()
    message(STATUS "KConfigCore unavailable: KDE session-ending actions disabled (authorization cannot be checked)")
endif()
if(KF6Service_FOUND)
    target_link_libraries(opm_native PUBLIC KF6::Service)
    target_compile_definitions(opm_native PRIVATE OPM_HAVE_KSERVICE=1)
endif()
target_compile_definitions(opm_native PRIVATE OPM_NATIVE_VERSION="${OPM_VERSION}")
target_include_directories(opm_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(opm_native PRIVATE -Wall -Wextra -Wpedantic)
add_executable(open-power-menu-native main.cpp service.cpp service.hpp)
target_link_libraries(open-power-menu-native PRIVATE opm_native Qt6::DBus)
target_compile_definitions(open-power-menu-native PRIVATE OPM_NATIVE_VERSION="${OPM_VERSION}")
install(TARGETS open-power-menu-native RUNTIME DESTINATION bin)

if(OPM_BUILD_PACKAGE)
    if(CMAKE_VERSION VERSION_LESS 3.22)
        message(FATAL_ERROR "Complete native bundles require CMake 3.22+ for distribution metadata")
    endif()
    if(BUILD_TESTING)
        add_executable(native-platform-plugin-tests tests/platform_plugins_test.cpp packaging/platform_plugins.cpp)
        target_link_libraries(native-platform-plugin-tests PRIVATE opm_native Qt6::Test)
        add_test(NAME native-platform-plugins COMMAND native-platform-plugin-tests)
    endif()
    add_executable(open-power-menu-install packaging/installer.cpp packaging/dependency_review.cpp packaging/platform_plugins.cpp)
    target_link_libraries(open-power-menu-install PRIVATE opm_native Qt6::DBus)
    target_compile_definitions(open-power-menu-install PRIVATE OPM_NATIVE_VERSION="${OPM_VERSION}" OPM_SOURCE_COMMIT="${OPM_SOURCE_COMMIT}")
    # Embed the exact released icon for conservative legacy ownership checks.
    execute_process(COMMAND base64 -w0 "${PROJECT_SOURCE_DIR}/assets/open-power-menu.svg" OUTPUT_VARIABLE OPM_ICON_BASE64 COMMAND_ERROR_IS_FATAL ANY)
    target_compile_definitions(open-power-menu-install PRIVATE OPM_ICON_BASE64="${OPM_ICON_BASE64}")
    if(OPM_INSTALL_TEST_HOOKS)
        target_compile_definitions(open-power-menu-install PRIVATE OPM_INSTALL_TEST_HOOKS=1)
    endif()
    add_executable(open-power-menu packaging/dispatcher.cpp)
    target_link_libraries(open-power-menu PRIVATE opm_native)
    install(TARGETS open-power-menu open-power-menu-install RUNTIME DESTINATION bin)
    cmake_host_system_information(RESULT OPM_PACKAGE_OS QUERY DISTRIB_ID)
    cmake_host_system_information(RESULT OPM_PACKAGE_OS_VERSION QUERY DISTRIB_VERSION_ID)
    configure_file(packaging/install.sh.in ${CMAKE_CURRENT_BINARY_DIR}/install.sh @ONLY)
    configure_file(packaging/uninstall.sh.in ${CMAKE_CURRENT_BINARY_DIR}/uninstall.sh @ONLY)
    configure_file(packaging/bundle.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/bundle.cmake @ONLY)
    configure_file(packaging/archive.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/archive.cmake @ONLY)
    add_custom_target(native-package
        COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_BINARY_DIR}/archive.cmake
        DEPENDS native-bundle)
    add_custom_target(native-bundle
        COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_BINARY_DIR}/bundle.cmake
        DEPENDS open-power-menu-native open-power-menu-editor open-power-menu-standalone open-power-menu-install open-power-menu opminputplugin)
endif()
