cmake_minimum_required(VERSION 3.21)
project(OpenPowerMenuNative VERSION 2.1.0 LANGUAGES CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/native/VERSION" OPM_VERSION LIMIT_COUNT 1)
if(NOT OPM_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-[0-9A-Za-z.]+)?$")
    message(FATAL_ERROR "Invalid native/VERSION")
endif()
if(NOT "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}" STREQUAL PROJECT_VERSION)
    message(FATAL_ERROR "native/VERSION and project version disagree")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/VERSION)
set(OPM_SOURCE_COMMIT "" CACHE STRING "Exact Git source commit for a distribution build")
if(NOT OPM_SOURCE_COMMIT)
    execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse HEAD
        OUTPUT_VARIABLE OPM_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core DBus)
include(CTest)
option(OPM_BUILD_EDITOR "Build the native Qt Quick/Kirigami configuration editor" OFF)
option(OPM_BUILD_STANDALONE "Build the native Qt Widgets standalone menu" OFF)
option(OPM_BUILD_PACKAGE "Build the complete native distribution bundle" OFF)
option(OPM_INSTALL_TEST_HOOKS "Enable installer crash injection in disposable tests only" OFF)
if(OPM_BUILD_PACKAGE)
    string(LENGTH "${OPM_SOURCE_COMMIT}" OPM_COMMIT_LENGTH)
    if(NOT OPM_COMMIT_LENGTH EQUAL 40 OR NOT OPM_SOURCE_COMMIT MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "Distribution builds require the exact 40-character OPM_SOURCE_COMMIT")
    endif()
    set(OPM_BUILD_EDITOR ON)
    set(OPM_BUILD_STANDALONE ON)
endif()
add_subdirectory(native)
