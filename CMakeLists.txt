cmake_minimum_required(VERSION 3.21)
project(OpenPowerMenuNative VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core DBus)
include(CTest)
option(OPM_BUILD_EDITOR "Build the native Qt Quick/Kirigami configuration editor" OFF)
option(OPM_BUILD_STANDALONE "Build the native Qt Widgets standalone menu" OFF)
option(OPM_BUILD_PACKAGE "Build the complete native distribution bundle" OFF)
option(OPM_INSTALL_TEST_HOOKS "Enable installer crash injection in disposable tests only" OFF)
if(OPM_BUILD_PACKAGE)
    set(OPM_BUILD_EDITOR ON)
    set(OPM_BUILD_STANDALONE ON)
endif()
add_subdirectory(native)
